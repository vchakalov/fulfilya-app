  -- PaymentSummaryReport - one row per day per payment method.
  --
  -- NOTE: prepared statements are turned OFF for this query (its settings tab) and must
  -- stay off while these comments exist - Appsmith refuses to run a prepared statement
  -- whose text contains the word "is" anywhere, including inside a comment.
  --
  -- Rewritten 2026-09-15. It had never returned a row, and could not have:
  --
  --   * it scoped on `o.company_uuid = <the merchant's CONTACT uuid>`. Those are two
  --     different identifiers, so the filter matched nothing. Every other query in the
  --     portal scopes on `o.customer_uuid`.
  --   * it read the COD amount from a custom field called `cod-amount:`, which does not
  --     exist in this deployment. The field is called `amount:`. Same fault that
  --     PaymentAnalysisReport
  --     was fixed for in 02a78a4.
  --   * it tested for statuses `delivered`, `cancelled` and `in_transit`, none of which
  --     this deployment ever writes. The real vocabulary: `completed` for a delivery,
  --     `order_cancel` for a cancellation, `dispatched` / `started` / `enroute` in the van.
  --   * its CTE selected `o.status` without grouping or aggregating it, which MySQL
  --     rejects outright under ONLY_FULL_GROUP_BY.
  --   * both ends of the date range came from StartDatePicker, so the end date was
  --     whatever the start picker happened to hold.
  --
  -- A returned parcel counts as neither collected nor lost: the money was never taken,
  -- and the goods came back. It gets its own column so the two numbers stay honest.
  WITH daily_orders AS (
    SELECT
      DATE(o.created_at) AS report_date,
      o.uuid,
      MIN(o.status) AS status,
      MAX(CASE WHEN cf.name = 'payment-method' THEN cfv.value END) AS payment_method,
      MAX(CASE WHEN cf.name = 'amount:' THEN REPLACE(REPLACE(cfv.value, '$', ''), ',', '') END) AS order_amount,
      MAX(EXISTS (
        SELECT 1 FROM activity a
        WHERE a.subject_id = o.uuid
          AND JSON_EXTRACT(a.properties, '$.attributes.status') = 'order_returned'
      )) AS was_returned
    FROM orders o
    LEFT JOIN custom_field_values cfv
      ON cfv.subject_uuid = o.uuid AND cfv.subject_type = 'Fleetbase\\FleetOps\\Models\\Order'
    LEFT JOIN custom_fields cf ON cfv.custom_field_uuid = cf.uuid
    WHERE o.customer_uuid = '{{appsmith.store.customer_uuid}}'
      AND o.deleted_at <=> NULL
      AND DATE(o.created_at) >= COALESCE(
        NULLIF('{{StartDatePicker.formattedDate}}', ''),
        DATE(NOW()) - INTERVAL 30 DAY
      )
      AND DATE(o.created_at) <= COALESCE(
        NULLIF('{{EndDatePicker.formattedDate}}', ''),
        DATE(NOW())
      )
    GROUP BY DATE(o.created_at), o.uuid
  )
  SELECT
    report_date,
    COALESCE(payment_method, 'Not Specified') AS payment_method,

    COUNT(*) AS total_orders,
    SUM(CASE WHEN status IN ('completed', 'order_delivered') AND was_returned = 0 THEN 1 ELSE 0 END) AS delivered,
    SUM(CASE WHEN was_returned = 1 THEN 1 ELSE 0 END) AS returned,
    SUM(CASE WHEN status IN ('order_cancel', 'canceled', 'cancelled') THEN 1 ELSE 0 END) AS cancelled,
    SUM(CASE WHEN status IN ('created', 'dispatched', 'started', 'enroute', 'delivery_attempted', 'returning') THEN 1 ELSE 0 END) AS in_progress,

    ROUND(
      SUM(CASE WHEN status IN ('completed', 'order_delivered') AND was_returned = 0 THEN 1 ELSE 0 END) * 100.0
      / NULLIF(COUNT(*), 0)
    , 1) AS delivery_rate,

    -- Cash actually taken at the door, cash still to be taken, and cash that never will be.
    ROUND(SUM(CASE
      WHEN payment_method = 'COD' AND status IN ('completed', 'order_delivered') AND was_returned = 0
      THEN CAST(order_amount AS DECIMAL(12,2)) ELSE 0
    END), 2) AS cod_collected,

    ROUND(SUM(CASE
      WHEN payment_method = 'COD' AND status IN ('created', 'dispatched', 'started', 'enroute', 'delivery_attempted', 'returning')
      THEN CAST(order_amount AS DECIMAL(12,2)) ELSE 0
    END), 2) AS cod_pending,

    ROUND(SUM(CASE
      WHEN payment_method = 'COD' AND (status IN ('order_cancel', 'canceled', 'cancelled') OR was_returned = 1)
      THEN CAST(order_amount AS DECIMAL(12,2)) ELSE 0
    END), 2) AS cod_not_collected

  FROM daily_orders
  GROUP BY report_date, COALESCE(payment_method, 'Not Specified')
  ORDER BY report_date DESC, payment_method;
