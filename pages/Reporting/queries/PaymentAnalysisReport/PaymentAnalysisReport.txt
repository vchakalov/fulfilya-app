  -- Payment analysis, one row per way an order was actually paid for:
  --   COD - Cash   customer paid the driver in notes
  --   COD - Card   customer paid on the driver's POS terminal (the `cod-with-card`
  --                custom field, uuid below, set by the driver app at the door)
  --   CARD         paid online before the order ever reached us
  --
  -- Statuses are this deployment's real vocabulary: `completed` is delivered,
  -- 'delivered' / 'in_transit' never appear. The order amount is the custom field
  -- literally named `amount:` - an earlier version of this query looked for
  -- `cod-amount:`, which does not exist, so every money column read zero.
  SELECT
    CASE
      WHEN COALESCE(cfd.payment_method, '') = '' THEN 'Not Specified'
      WHEN cfd.payment_method <> 'COD' THEN cfd.payment_method
      WHEN cfd.paid_with_card = 'TRUE' THEN 'COD - Card'
      ELSE 'COD - Cash'
    END as payment_method,

    COUNT(DISTINCT o.uuid) as total_orders,
    COUNT(DISTINCT CASE WHEN o.status = 'completed' THEN o.uuid END) as delivered_orders,
    COUNT(DISTINCT CASE WHEN o.status IN ('canceled', 'cancelled', 'order_cancel') THEN o.uuid END) as cancelled_orders,
    COUNT(DISTINCT CASE WHEN o.status IN ('created', 'dispatched', 'enroute', 'driver_enroute', 'started') THEN o.uuid END) as in_progress_orders,

    -- Success Rates
    ROUND(COUNT(DISTINCT CASE WHEN o.status = 'completed' THEN o.uuid END) * 100.0 / NULLIF(COUNT(DISTINCT o.uuid), 0), 1) as delivery_success_rate,
    ROUND(COUNT(DISTINCT CASE WHEN o.status IN ('canceled', 'cancelled', 'order_cancel') THEN o.uuid END) * 100.0 / NULLIF(COUNT(DISTINCT o.uuid), 0), 1) as cancellation_rate,

    -- Financial Analysis
    ROUND(SUM(CASE WHEN o.status = 'completed' THEN cfd.amount ELSE 0 END), 2) as total_revenue,
    ROUND(SUM(CASE WHEN o.status IN ('created', 'dispatched', 'enroute', 'driver_enroute', 'started') THEN cfd.amount ELSE 0 END), 2) as pending_revenue,
    ROUND(AVG(NULLIF(cfd.amount, 0)), 2) as avg_order_value

  FROM orders o
  LEFT JOIN (
    SELECT
      cfv.subject_uuid,
      MAX(CASE WHEN cf.name = 'payment-method' THEN cfv.value END) as payment_method,
      MAX(CASE WHEN cfv.custom_field_uuid = '38c677ff-5bd4-4087-bb0e-2f4c031153c7' THEN cfv.value END) as paid_with_card,
      CAST(COALESCE(MAX(CASE WHEN cf.name = 'amount:'
        THEN REPLACE(REPLACE(REPLACE(cfv.value, '$', ''), '€', ''), ',', '') END), '0') AS DECIMAL(10,2)) as amount
    FROM custom_field_values cfv
    LEFT JOIN custom_fields cf ON cfv.custom_field_uuid = cf.uuid
    WHERE cfv.subject_type = 'Fleetbase\\FleetOps\\Models\\Order'
      AND cfv.deleted_at IS NULL
    GROUP BY cfv.subject_uuid
  ) cfd ON cfd.subject_uuid = o.uuid

  WHERE o.customer_uuid = '{{appsmith.store.customer_uuid}}'
    AND o.created_at >= '{{moment(StartDatePicker.selectedDate).format("YYYY-MM-DD")}}'
    AND o.created_at <= '{{moment(EndDatePicker.selectedDate).format("YYYY-MM-DD")}} 23:59:59'
  GROUP BY 1
  ORDER BY total_orders DESC;
