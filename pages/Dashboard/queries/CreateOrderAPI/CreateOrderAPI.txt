{{(() => {
	// Everything about WHO this order belongs to is resolved on the server from the
	// merchant's own login token, so this body carries only what the merchant typed.
	// It can no longer name another merchant, pick another merchant's price, or point
	// at a different order config - those used to travel from the browser, signed with
	// an admin token belonging to the company owner.
	//
	// The delivery quote and the наложен платеж commission are worked out server-side
	// too, which is why CreateServiceQuoteForOrder and GetCreatedServiceQuote are no
	// longer run before this: their SQL reused one quote row for a whole day of orders
	// and never linked it to the order, so nothing the merchant is paid out on could
	// be traced back to a price.

	if (!NewDropoffAddressInput.text) {
		throw new Error("Адресът за доставка е задължителен");
	}

	if (!OrderShippingSelect.selectedOptionValue) {
		throw new Error("Изберете вид доставка");
	}

	// The recipient. Until 2026-09-19 this form had no field for either, so every order it
	// created reached the driver with a pin and nobody to ring at the door - the shop
	// webhook has always sent both. Refused here rather than left blank: a courier cannot
	// work an address without a phone.
	const NewRecipientName = (RecipientNameInput.text || "").trim();
	const NewRecipientPhone = (RecipientPhoneInput.text || "").trim();

	if (!NewRecipientName) {
		throw new Error("Въведете име на получателя");
	}

	if (!NewRecipientPhone) {
		throw new Error("Въведете телефон на получателя");
	}

	const NewCustomerEmail = (CustomerEmailInput.text || "").trim();

	return {
		// A single line the server geocodes against Bulgaria, which is what gives the
		// driver a real street, city and postcode. The old call sent coordinates and a
		// formatted string that Fleetbase silently dropped.
		"dropoff": {
			"address": NewDropoffAddressInput.text,
			"name": NewRecipientName,
			"phone": NewRecipientPhone,
			// The shopper's own email, so the office can write to them about the
			// delivery. Optional, and left out entirely when blank rather than sent
			// empty - the server validates it as an address when it arrives.
			...(NewCustomerEmail ? { "email": NewCustomerEmail } : {})
		},
		"service_rate_uuid": OrderShippingSelect.selectedOptionValue,
		"payment_method": NewPaymentMethodSelect.selectedOptionValue,
		"amount": NewAmountInput.text,
		"store_type": StoreTypeSelect.selectedOptionValue,
		"order_source_id": OrderSourceIdInput.text || "",
		"scheduled_at": ScheduledDeliveryPicker.selectedDate,
		"notes": DeliveryNotesInput.text || "",
		"items": (appsmith.store.newOrderItems || []).map(item => ({
			"name": item.name,
			// `ski` is the key the Add-item button has always written; read both so a
			// fixed button and an unfixed one both work.
			"sku": item.sku || item.ski || "",
			"quantity": item.quantity || 1
		}))
	};
})()}}
